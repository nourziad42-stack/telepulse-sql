Phase 3 - SQL DDL Code (Microsoft SQL Server)

-- Create Database
CREATE DATABASE TelePulseCRM;

GO
USE TelePulseCRM;
GO




-- Table 1: Department
CREATE TABLE Department (
    DepartmentID   INT NOT NULL PRIMARY KEY,
    DepartmentName VARCHAR(100) NOT NULL UNIQUE,
    Description    VARCHAR(255)
);

-- Table 2: Shift
CREATE TABLE Shift (
    ShiftID   INT NOT NULL PRIMARY KEY,
    ShiftName VARCHAR(50) NOT NULL UNIQUE,
    StartTime TIME NOT NULL,
    EndTime   TIME NOT NULL
);

-- Table 3: Employee
CREATE TABLE Employee (
    EmployeeID   INT NOT NULL PRIMARY KEY,
    FullName     VARCHAR(100) NOT NULL,
    Email        VARCHAR(120) NOT NULL UNIQUE,
    PasswordHash VARCHAR(255) NOT NULL,
    Role         VARCHAR(30) NOT NULL CHECK (Role IN ('Agent', 'Supervisor', 'Campaign Manager', 'Admin')),
    Status       VARCHAR(20) NOT NULL DEFAULT 'Active' CHECK (Status IN ('Active', 'Inactive')),
    HireDate     DATE NOT NULL,
    DepartmentID INT NOT NULL,
    ShiftID      INT NOT NULL,
    SupervisorID INT,
    FOREIGN KEY (DepartmentID) REFERENCES Department (DepartmentID) ON UPDATE CASCADE,
    FOREIGN KEY (ShiftID) REFERENCES Shift (ShiftID) ON UPDATE CASCADE,
    FOREIGN KEY (SupervisorID) REFERENCES Employee (EmployeeID)
);

-- Table 4: Campaign
CREATE TABLE Campaign (
    CampaignID        INT NOT NULL PRIMARY KEY,
    CampaignName      VARCHAR(120) NOT NULL,
    StartDate         DATE NOT NULL,
    EndDate           DATE NOT NULL,
    Budget            DECIMAL(10, 2) NOT NULL CHECK (Budget >= 0),
    TargetConversions INT NOT NULL CHECK (TargetConversions >= 0),
    Status            VARCHAR(20) NOT NULL DEFAULT 'Planned'
                      CHECK (Status IN ('Planned', 'Active', 'Paused', 'Completed', 'Cancelled')),
    ManagerID         INT NOT NULL,
    CHECK (EndDate >= StartDate),
    FOREIGN KEY (ManagerID) REFERENCES Employee (EmployeeID) ON UPDATE CASCADE
);

-- Table 5: Call_Script
CREATE TABLE Call_Script (
    ScriptID   INT NOT NULL PRIMARY KEY,
    Title      VARCHAR(120) NOT NULL,
    Content    VARCHAR(MAX) NOT NULL,
    Version    VARCHAR(20) NOT NULL DEFAULT '1.0',
    CampaignID INT NOT NULL,
    FOREIGN KEY (CampaignID) REFERENCES Campaign (CampaignID) ON DELETE CASCADE ON UPDATE CASCADE
);

-- Table 6: Lead
CREATE TABLE Lead (
    LeadID          INT NOT NULL PRIMARY KEY,
    FullName        VARCHAR(100) NOT NULL,
    Phone           VARCHAR(25) NOT NULL UNIQUE,
    Email           VARCHAR(120),
    Source          VARCHAR(80),
    LeadStatus      VARCHAR(30) NOT NULL DEFAULT 'New'
                    CHECK (LeadStatus IN ('New', 'Contacted', 'Interested', 'Callback', 'Converted', 'Do Not Call')),
    AssignedAgentID INT,
    AssignedDate    DATE,
    FOREIGN KEY (AssignedAgentID) REFERENCES Employee (EmployeeID) ON DELETE SET NULL ON UPDATE CASCADE
);

-- Table 7: Campaign_Lead
CREATE TABLE Campaign_Lead (
    CampaignID       INT NOT NULL,
    LeadID           INT NOT NULL,
    AssignedDate     DATE NOT NULL DEFAULT GETDATE(),
    UnassignedDate   DATE,
    AssignmentStatus VARCHAR(20) NOT NULL DEFAULT 'Active'
                     CHECK (AssignmentStatus IN ('Active', 'Completed', 'Removed')),
    PRIMARY KEY (CampaignID, LeadID),
    FOREIGN KEY (CampaignID) REFERENCES Campaign (CampaignID) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (LeadID) REFERENCES Lead (LeadID) ON DELETE CASCADE
);

-- Table 8: Call_Log
CREATE TABLE Call_Log (
    CallID          INT NOT NULL PRIMARY KEY,
    CallDateTime    DATETIME NOT NULL DEFAULT GETDATE(),
    DurationSeconds INT NOT NULL CHECK (DurationSeconds >= 0),
    CallResult      VARCHAR(30) NOT NULL
                    CHECK (CallResult IN ('No Answer', 'Busy', 'Interested', 'Callback', 'Converted', 'Not Interested', 'Do Not Call')),
    Notes           VARCHAR(500),
    LeadID          INT NOT NULL,
    AgentID         INT NOT NULL,
    FOREIGN KEY (LeadID) REFERENCES Lead (LeadID) ON UPDATE CASCADE,
    FOREIGN KEY (AgentID) REFERENCES Employee (EmployeeID)
);

-- Table 9: Follow_Up_Task
CREATE TABLE Follow_Up_Task (
    TaskID          INT NOT NULL PRIMARY KEY,
    DueDateTime     DATETIME NOT NULL,
    TaskStatus      VARCHAR(20) NOT NULL DEFAULT 'Pending'
                    CHECK (TaskStatus IN ('Pending', 'In Progress', 'Completed', 'Overdue', 'Cancelled')),
    Notes           VARCHAR(500),
    CallID          INT NOT NULL,
    AssignedAgentID INT NOT NULL,
    FOREIGN KEY (CallID) REFERENCES Call_Log (CallID) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (AssignedAgentID) REFERENCES Employee (EmployeeID)
);

-- Table 10: Quality_Evaluation
CREATE TABLE Quality_Evaluation (
    EvaluationID   INT NOT NULL PRIMARY KEY,
    Score          INT NOT NULL CHECK (Score BETWEEN 0 AND 100),
    EvaluationDate DATETIME NOT NULL DEFAULT GETDATE(),
    Comments       VARCHAR(500),
    CallID         INT NOT NULL,
    SupervisorID   INT NOT NULL,
    FOREIGN KEY (CallID) REFERENCES Call_Log (CallID) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (SupervisorID) REFERENCES Employee (EmployeeID)
);

-- Table 11: Product_Category
CREATE TABLE Product_Category (
    CategoryID   INT NOT NULL PRIMARY KEY,
    CategoryName VARCHAR(100) NOT NULL UNIQUE,
    Description  VARCHAR(255)
);

-- Table 12: Product
CREATE TABLE Product (
    ProductID   INT NOT NULL PRIMARY KEY,
    ProductName VARCHAR(120) NOT NULL,
    Description VARCHAR(255),
    UnitPrice   DECIMAL(10, 2) NOT NULL CHECK (UnitPrice >= 0),
    CategoryID  INT NOT NULL,
    FOREIGN KEY (CategoryID) REFERENCES Product_Category (CategoryID) ON UPDATE CASCADE
);

-- Table 13: Sales_Order
CREATE TABLE Sales_Order (
    OrderID     INT NOT NULL PRIMARY KEY,
    OrderDate   DATETIME NOT NULL DEFAULT GETDATE(),
    OrderStatus VARCHAR(20) NOT NULL DEFAULT 'Pending'
                CHECK (OrderStatus IN ('Pending', 'Confirmed', 'Completed', 'Cancelled')),
    LeadID      INT NOT NULL,
    AgentID     INT NOT NULL,
    FOREIGN KEY (LeadID) REFERENCES Lead (LeadID) ON UPDATE CASCADE,
    FOREIGN KEY (AgentID) REFERENCES Employee (EmployeeID)
);

-- Table 14: Order_Item (Weak Entity with Composite Key OrderID + ItemNo)
CREATE TABLE Order_Item (
    OrderID   INT NOT NULL,
    ItemNo    INT NOT NULL CHECK (ItemNo > 0),
    ProductID INT NOT NULL,
    Quantity  INT NOT NULL CHECK (Quantity > 0),
    UnitPrice DECIMAL(10, 2) NOT NULL CHECK (UnitPrice >= 0),
    Discount  DECIMAL(5, 2) NOT NULL DEFAULT 0.00 CHECK (Discount BETWEEN 0 AND 100),
    PRIMARY KEY (OrderID, ItemNo),
    FOREIGN KEY (OrderID) REFERENCES Sales_Order (OrderID) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (ProductID) REFERENCES Product (ProductID) ON UPDATE CASCADE
);

-- Table 15: Invoice (1:1 with Sales_Order -> OrderID is UNIQUE)
CREATE TABLE Invoice (
    InvoiceID     INT NOT NULL PRIMARY KEY,
    InvoiceDate   DATE NOT NULL DEFAULT GETDATE(),
    DueDate       DATE NOT NULL,
    TotalAmount   DECIMAL(10, 2) NOT NULL CHECK (TotalAmount >= 0),
    InvoiceStatus VARCHAR(20) NOT NULL DEFAULT 'Unpaid'
                  CHECK (InvoiceStatus IN ('Unpaid', 'Partially Paid', 'Paid', 'Overdue', 'Cancelled')),
    OrderID       INT NOT NULL UNIQUE,
    CHECK (DueDate >= InvoiceDate),
    FOREIGN KEY (OrderID) REFERENCES Sales_Order (OrderID) ON UPDATE CASCADE
);

-- Table 16: Payment
CREATE TABLE Payment (
    PaymentID     INT NOT NULL PRIMARY KEY,
    PaymentDate   DATETIME NOT NULL DEFAULT GETDATE(),
    Amount        DECIMAL(10, 2) NOT NULL CHECK (Amount > 0),
    PaymentMethod VARCHAR(30) NOT NULL
                  CHECK (PaymentMethod IN ('Cash', 'Credit Card', 'Debit Card', 'Bank Transfer', 'Mobile Wallet')),
    PaymentStatus VARCHAR(20) NOT NULL DEFAULT 'Completed'
                  CHECK (PaymentStatus IN ('Pending', 'Completed', 'Failed', 'Refunded')),
    InvoiceID     INT NOT NULL,
    FOREIGN KEY (InvoiceID) REFERENCES Invoice (InvoiceID) ON DELETE CASCADE ON UPDATE CASCADE
);
